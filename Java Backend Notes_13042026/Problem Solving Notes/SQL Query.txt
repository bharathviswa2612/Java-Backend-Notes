SQL Query 
----------

1.Find the second highest salary

SELECT salary FROM Employee ORDER BY salary DESC LIMIT 1 OFFSET 1

2.Find the duplicate records 

SELECT name , COUNT(*) FROM Employee GROUP BY name HAVING COUNT(*) > 1

3.Delete Duplicate Records

DELETE e1 FROM Employee e1 JOIN Employee e2 ON e1.name = e2.name AND e1.id > e2.id

4.Count Employee per department 

SELECT d1.dep_name , COUNT(*) AS Empname FROM Employee e1 INNER JOIN Department d1 ON 
e1.dep_id = d1.dep_id GROUP BY d1.dep_name;

5.Employees with Salary > Average

SELECT * FROM Employee WHERE salary > (SELECT AVG(salary) FROM Employee)

6.Find the total number of orders per customer Id ?

SELECT customer_id , COUNT(*) FROM Orders GROUP BY customer_id

7.Find the employee who joined in the last 30 days ?

SELECT * FROM Employee WHERE Joining_date >= CURDATE() - INTERVAL 30 DAY 

8.Find the 6th Highest salary ?

SELECT salary FROM Employee ORDER BY salary DESC LIMIT 1 OFFSET 5;

9.Find the Maximum salary from each department 

SELECT dep_name , MAX(salary) FROM Employee GROUP BY dep_name;

10.Find the Employee who joined between 01-03-2026 and 10-03-2026 ?

SELECT * FROM Employee WHERE Joining_date BETWEEN '01-03-2026' AND '10-03-2026'
(or)
SELECT * FROM Employee WHERE joining_date >= '2026-03-01' AND joining_date < '2026-03-11';

11.Find departments where more than 3 employees earn salary greater than 60,000 ?

SELECT department , COUNT(*) FROM employees
WHERE salary > 60000
GROUP BY department
HAVING COUNT(*) > 3

12.Find departments where more than 4 employees earn salary greater than 50,000.

SELECT dept_name , COUNT(*) AS EmpCount FROM Employee WHERE salary > 50000
GROUP BY dept_name HAVING COUNT(*) > 4;
